import java.util.Scanner;

public class IT26102586Lab3Q2{
         
		public static void main(String[] args){
		double monthlysalary;
		double OTamount;
		double OThours=5;
        double OThourlyRate=1000.0;
		double totalSalary;
		
		//create object
		Scanner input=new Scanner(System.in);
		
		System.out.println("Enter the monthly salary");
		monthlysalary=input.nextDouble();
		
		System.out.println("Enter the OT hours");
		OThours=input.nextDouble();
		
		OTamount=OThours*OThourlyRate;
		
		totalSalary=monthlysalary+OTamount;
		
		System.out.println("The totalSalary:"+totalSalary);
		System.out.println("The monthlysalary:"+monthlysalary);
		System.out.println("The OTamount:"+OTamount);
		
		
		}
		
		
		
}
