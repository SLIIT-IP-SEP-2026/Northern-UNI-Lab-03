import java.util.Scanner;

public class IT26102586Lab3Q4{
         
		public static void main(String[] args){
			int number;
			int d1,d2,d3,d4,d5;
			
			Scanner input=new Scanner(System.in);
			
			System.out.println("Enter the five-digital number:");
			number=input.nextInt();
			
			d1=number/10000;
			d2=(number/1000)%10;
			d3=(number/100)%10;
			d4=(number/10)%10;
			d5=number%10;
			
			System.out.println(d1+" "+d2+" "+d3+" "+d4+" "+d5);
			
		}
		
}
